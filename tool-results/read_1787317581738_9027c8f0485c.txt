    46→# ─── LRU CACHE ────────────────────────────────────────────────────────────────
    47→class LRUCache:
    48→    def __init__(self, maxsize=MAX_GAMES_IN_MEMORY):
    49→        self._cache = OrderedDict()
    50→        self._maxsize = maxsize
    51→
    52→    def get(self, key):
    53→        if key in self._cache:
    54→            self._cache.move_to_end(key)
    55→            return self._cache[key]
    56→        return None
    57→
    58→    def set(self, key, value):
    59→        if key in self._cache:
    60→            self._cache.move_to_end(key)
    61→        self._cache[key] = value
    62→        while len(self._cache) > self._maxsize:
    63→            self._cache.popitem(last=False)
    64→
    65→    def clear(self):
    66→        self._cache.clear()

   122→def implied_prob(odds_decimal):
   123→    if not odds_decimal or odds_decimal <= 1.01:
   124→        return 1.0 if odds_decimal and odds_decimal <= 1.01 else 0.0

   133→def get_elapsed_sec(game):
   134→    t = game.get("time", {})
   135→    try:
   136→        q = int(t.get("q", 0))
   137→        tm = int(t.get("tm", 0))
   138→        ts = int(t.get("ts", 0))
   139→        qt = t.get("qTime", 10)
   140→        elapsed = (q - 1) * qt * 60 + (qt * 60 - tm * 60 - ts)
   141→        return max(0, min(elapsed, 2400))
   142→    except (ValueError, TypeError):
   143→        return 0

   165→# ─── PREDICTION ENGINE ────────────────────────────────────────────────────────
   166→def predict_moneyline(markets):
   167→    ml = markets.get("1X2")
   168→    if not ml or not ml["opening"] or not ml["latest"]:
   169→        return None
   170→    op, cur = ml["opening"], ml["latest"]
   171→    op_h, op_a = op.get("row1"), op.get("row3")
   172→    cur_h, cur_a = cur.get("row1"), cur.get("row3")
   173→    if not all([op_h, op_a, cur_h, cur_a]):
   174→        return None
   175→    if cur_h < 1.05 or cur_a < 1.05:
   176→        return None
   177→
   178→    op_hp, op_ap = implied_prob(op_h), implied_prob(op_a)
   179→    cur_hp, cur_ap = implied_prob(cur_h), implied_prob(cur_a)
   180→    h_shift = cur_hp - op_hp
   181→    a_shift = cur_ap - op_ap
   182→
   183→    if abs(h_shift) < 0.03 and abs(a_shift) < 0.03:
   184→        return None
   185→
   186→    if h_shift > a_shift:
   187→        pick, shift_pct = "HOME", h_shift * 100
   188→    else:
   189→        pick, shift_pct = "AWAY", a_shift * 100
   190→
   191→    raw_conf = min(95, 30 + abs(shift_pct) * 2.5)
   192→
   193→    # Cross-validate with Handicap rating
   194→    rating_boost = 0
   195→    hc = markets.get("Handicap")
   196→    if hc and hc["latest"]:
   197→        for r in hc["latest"].get("rating", []):
   198→            if r and isinstance(r, dict):
   199→                rd = r.get("direction", "")
   200→                rv = abs(r.get("rating", 0))
   201→                if (pick == "HOME" and rd == "Home") or (pick == "AWAY" and rd == "Away"):
   202→                    rating_boost = min(10, rv * 2)
   203→                    break
   204→
   205→    # Also cross-validate with HT 1X2 direction
   206→    ht_ml = markets.get("HT 1X2")
   207→    if ht_ml and ht_ml["latest"]:
   208→        ht_op, ht_cur = ht_ml["opening"], ht_ml["latest"]
   209→        ht_op_h, ht_op_a = ht_op.get("row1"), ht_op.get("row3")
   210→        ht_cur_h, ht_cur_a = ht_cur.get("row1"), ht_cur.get("row3")
   211→        if all([ht_op_h, ht_op_a, ht_cur_h, ht_cur_a]):
   212→            ht_h_shift = implied_prob(ht_cur_h) - implied_prob(ht_op_h)
   213→            ht_a_shift = implied_prob(ht_cur_a) - implied_prob(ht_op_a)
   214→            ht_pick = "HOME" if ht_h_shift > ht_a_shift else "AWAY"
   215→            if ht_pick == pick:
   216→                rating_boost = min(rating_boost + 5, 12)
   217→
   218→    confidence = min(98, raw_conf + rating_boost)
   219→    details = {
   220→        "op_home": op_h, "op_away": op_a,
   221→        "cur_home": cur_h, "cur_away": cur_a,
   222→        "home_shift": round(h_shift * 100, 1),
   223→        "away_shift": round(a_shift * 100, 1),
   224→        "rating_boost": round(rating_boost, 1),
   225→    }
   226→    return (pick, round(confidence, 1), details)

   229→def predict_total(markets, game):
   230→    tot = markets.get("Total")
   231→    if not tot or not tot["opening"] or not tot["latest"]:
   232→        return None
   233→    op, cur = tot["opening"], tot["latest"]
   234→    op_line, cur_line = op.get("row2"), cur.get("row2")
   235→    cur_over, cur_under = cur.get("row1"), cur.get("row3")
   236→    if not all([op_line, cur_line, cur_over, cur_under]):
   237→        return None
   238→
   239→    line_move = cur_line - op_line
   240→    hs, as_ = get_scores(game)
   241→    total_score = hs + as_
   242→    elapsed = get_elapsed_sec(game)
   243→    full_time = tot.get("full_time", 2400)
   244→
   245→    if elapsed < 300:
   246→        return None
   247→
   248→    projected = (total_score / elapsed) * full_time
   249→    pace_gap = projected - cur_line
   250→
   251→    over_prob, under_prob = implied_prob(cur_over), implied_prob(cur_under)
   252→
   253→    score_sig = 1 if pace_gap > 3 else (-1 if pace_gap < -3 else 0)
   254→    line_sig = -1 if line_move < -2 else (1 if line_move > 2 else 0)
   255→    odds_sig = 1 if over_prob > under_prob else -1
   256→    combined = score_sig * 3 + line_sig * 2 + odds_sig * 1
   257→
   258→    if combined == 0:
   259→        return None
   260→
   261→    pick = "OVER" if combined > 0 else "UNDER"
   262→
   263→    raw_conf = 35
   264→    raw_conf += min(25, abs(pace_gap) * 2)
   265→    raw_conf += min(20, abs(line_move) * 3)
   266→    raw_conf += min(10, abs(combined) * 3)
   267→
   268→    rating_boost = 0
   269→    for r in cur.get("rating", []):
   270→        if r and isinstance(r, dict):
   271→            rd = r.get("direction", "")
   272→            if (pick == "OVER" and rd == "Over") or (pick == "UNDER" and rd == "Under"):
   273→                rating_boost = min(10, abs(r.get("rating", 0)) * 3)
   274→                break
   275→            else:
   276→                rating_boost = -5
   277→
   278→    # Cross-validate with HT Total direction
   279→    ht_tot = markets.get("HT Total")
   280→    if ht_tot and ht_tot["latest"]:
   281→        ht_op_line = ht_tot["opening"].get("row2") if ht_tot["opening"] else None
   282→        ht_cur_line = ht_tot["latest"].get("row2")
   283→        if ht_op_line and ht_cur_line:
   284→            ht_line_move = ht_cur_line - ht_op_line
   285→            if (pick == "OVER" and ht_line_move > 1) or (pick == "UNDER" and ht_line_move < -1):
   286→                rating_boost = min(rating_boost + 4, 12)
   287→
   288→    confidence = max(30, min(95, raw_conf + rating_boost))
   289→    details = {
   290→        "op_line": op_line, "cur_line": cur_line,
   291→        "line_move": round(line_move, 1),
   292→        "current_score": total_score,
   293→        "elapsed_min": round(elapsed / 60, 1),
   294→        "projected_total": round(projected, 1),
   295→        "pace_gap": round(pace_gap, 1),
   296→        "cur_over_odds": cur_over, "cur_under_odds": cur_under,
   297→        "rating_boost": rating_boost,
   298→    }
   299→    return (pick, round(confidence, 1), cur_line, details)

   302→def analyze_game(game, odds_data):
   303→    markets = parse_markets(odds_data)
   304→    preds = []
   305→    if get_quarter(game) < 2:
   306→        return preds
   307→
   308→    ml = predict_moneyline(markets)
   309→    if ml:
   310→        pick, conf, details = ml
   311→        if conf >= CONFIDENCE_THRESHOLD:
   312→            preds.append({"type": "MONEYLINE", "pick": pick, "confidence": conf, "details": details})
   313→
   314→    tot = predict_total(markets, game)
   315→    if tot:
   316→        pick, conf, line, details = tot
   317→        if conf >= CONFIDENCE_THRESHOLD:
   318→            preds.append({"type": "TOTAL", "pick": pick, "line": line, "confidence": conf, "details": details})
   319→
   320→    preds.sort(key=lambda x: x["confidence"], reverse=True)
   321→    return preds, markets

   324→# ─── MESSAGE FORMATTING ───────────────────────────────────────────────────────
   325→def fmt_pred(game, pred, rank=1):
   326→    home = game["home"]["name"]
   327→    away = game["away"]["name"]
   328→    league = game["league"]["name"]
   329→    icon = "\U0001f525" if rank == 1 else "\u26a1"
   330→    lines = [
   331→        f"{icon} #{rank} {pred['type']}",
   332→        f"{home} vs {away}",
   333→        f"{league} | {fmt_time(game)} | {game.get('scores', '0-0')}",
   334→        f"Confidence: {pred['confidence']}%",
   335→        "-" * 30,
   336→    ]
   337→    d = pred["details"]
   338→    if pred["type"] == "MONEYLINE":
   339→        lines.append(f"Pick: <b>{pred['pick']}</b>")
   340→        lines.append(f"Opening: H {d['op_home']} / A {d['op_away']}")
   341→        lines.append(f"Current: H {d['cur_home']} / A {d['cur_away']}")
   342→        hs = f"+{d['home_shift']}%" if d['home_shift'] > 0 else f"{d['home_shift']}%"
   343→        as_ = f"+{d['away_shift']}%" if d['away_shift'] > 0 else f"{d['away_shift']}%"
   344→        lines.append(f"Shift: Home {hs} | Away {as_}")
   345→        rb = round(d['rating_boost'], 1)
   346→        if rb > 0:
   347→            lines.append(f"Rating confirm: +{rb}%")
   348→    else:
   349→        lines.append(f"Pick: <b>{pred['pick']} {pred['line']}</b>")
   350→        lines.append(f"Line: {d['op_line']} -> {d['cur_line']} ({d['line_move']:+.1f})")
   351→        lines.append(f"Score: {d['current_score']}pts in {d['elapsed_min']}min")
   352→        lines.append(f"Pace: {d['projected_total']}pts (gap {d['pace_gap']:+.1f})")
   353→        lines.append(f"Odds: Over {d['cur_over_odds']} / Under {d['cur_under_odds']}")
   354→        rb = round(d['rating_boost'], 1)
   355→        if rb > 0:
   356→            lines.append(f"Rating confirm: +{rb}%")
   357→        elif rb < 0:
   358→            lines.append(f"Rating disagree: {rb}%")
   359→    return "\n".join(lines), game["id"]

   420→# ─── SCAN ENGINE ──────────────────────────────────────────────────────────────
   421→def scan_and_predict():
   422→    global scan_count, pred_count
   423→    games = fetch_live_games()
   424→    if not games:
   425→        return None, None, [], games
   426→    scan_count += 1
   427→    eligible = [g for g in games if get_quarter(g) >= 2]
   428→    if not eligible:
   429→        return None, None, [], games
   430→
   431→    results = []
   432→    for g in eligible:
   433→        try:
   434→            odds = fetch_odds(g["id"])
   435→            if not odds:
   436→                continue
   437→            preds, markets = analyze_game(g, odds)
   438→            for p in preds:
   439→                results.append((g, p))
   440→                pred_count += 1
   441→        except Exception:
   442→            pass
   443→
   444→    results.sort(key=lambda x: x[1]["confidence"], reverse=True)
   445→    if results:
   446→        return results[0][0], results[0][1], results, games
   447→    return None, None, [], games

   450→# ─── AUTO-SCAN BACKGROUND WORKER ─────────────────────────────────────────
   451→def auto_scan_worker():
   452→    """Fully automatic background scanner. Runs forever, pushes predictions."""
   453→    global auto_tracking, known_games
   454→    auto_tracking = True
   455→    cycle = 0
   456→
   457→    while auto_tracking:
   458→        cycle += 1
   459→        try:
   460→            bg, bp, all_results, games = scan_and_predict()
   461→
   462→            # --- New game alerts: notify when games enter Q2 ---
   463→            if games and NEW_GAME_ALERT:
   464→                current_ids = {g["id"] for g in games if get_quarter(g) >= 2}
   465→                new_ids = current_ids - known_games
   466→                if new_ids:
   467→                    new_games = [g for g in games if g["id"] in new_ids]
   468→                    for ng in new_games[:5]:
   469→                        bot.send_message(CHAT_ID,
   470→                            f"\U0001f3c0 <b>New game entered Q2+</b>\n"
   471→                            f"{ng['home']['name']} vs {ng['away']['name']}\n"
   472→                            f"{ng['league']['name']} | {ng.get('scores', '0-0')}\n"
   473→                            f"\U0001f4c5 ID: <code>{ng['id']}</code>",
   474→                            parse_mode="HTML"
   475→                        )
   476→                known_games = current_ids
   477→
   478→            # --- Send predictions ---
   479→            if bg and bp:
   480→                if ALL_PREDICTIONS and all_results:
   481→                    for g, p in all_results:
   482→                        key = f"{g['id']}_{p['type']}"
   483→                        old = last_predictions.get(key, 0)
   484→                        new_conf = p["confidence"]
   485→                        if abs(new_conf - old) >= 5:
   486→                            rank = 1 if (g, p) == (bg, bp) else 2
   487→                            send_pred_with_button(g, p, rank=rank)
   488→                            last_predictions[key] = new_conf
   489→                else:
   490→                    key = f"{bg['id']}_{bp['type']}"
   491→                    old = last_predictions.get(key, 0)
   492→                    new_conf = bp["confidence"]
   493→                    if abs(new_conf - old) >= 5:
   494→                        send_pred_with_button(bg, bp, rank=1)
   495→                        last_predictions[key] = new_conf
   496→
   497→            # Periodic status every 10 cycles (10 min)
   498→            if cycle % 10 == 0 and games:
   499→                eligible = len([g for g in games if get_quarter(g) >= 2])
   500→                bot.send_message(CHAT_ID,
   501→                    f"\U0001f4ca Scan #{scan_count} | {len(games)} live | {eligible} analyzing | {pred_count} predictions sent",
   502→                    parse_mode="HTML"
