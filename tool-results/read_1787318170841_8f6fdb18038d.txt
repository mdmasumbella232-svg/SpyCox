    47→    47→class LRUCache:
    48→    48→    def __init__(self, maxsize=MAX_GAMES_IN_MEMORY):
    49→    49→        self._cache = OrderedDict()
    50→    50→        self._maxsize = maxsize
    51→    51→
    52→    52→    def get(self, key):
    53→    53→        if key in self._cache:
    54→    54→            self._cache.move_to_end(key)
    55→    55→            return self._cache[key]
    56→    56→        return None
    57→    57→
    58→    58→    def set(self, key, value):
    59→    59→        if key in self._cache:
    60→    60→            self._cache.move_to_end(key)
    61→    61→        self._cache[key] = value
    62→    62→        while len(self._cache) > self._maxsize:
    63→    63→            self._cache.popitem(last=False)
    64→    64→
    65→    65→    def clear(self):
    66→    66→        self._cache.clear()

   134→   134→    t = game.get("time", {})

   136→   136→        q = int(t.get("q", 0))

   139→   139→        qt = t.get("qTime", 10)

   165→   165→# ─── PREDICTION ENGINE ────────────────────────────────────────────────────────
   166→   166→def predict_moneyline(markets):
   167→   167→    ml = markets.get("1X2")
   168→   168→    if not ml or not ml["opening"] or not ml["latest"]:
   169→   169→        return None
   170→   170→    op, cur = ml["opening"], ml["latest"]
   171→   171→    op_h, op_a = op.get("row1"), op.get("row3")
   172→   172→    cur_h, cur_a = cur.get("row1"), cur.get("row3")
   173→   173→    if not all([op_h, op_a, cur_h, cur_a]):
   174→   174→        return None
   175→   175→    if cur_h < 1.05 or cur_a < 1.05:
   176→   176→        return None
   177→   177→
   178→   178→    op_hp, op_ap = implied_prob(op_h), implied_prob(op_a)
   179→   179→    cur_hp, cur_ap = implied_prob(cur_h), implied_prob(cur_a)
   180→   180→    h_shift = cur_hp - op_hp
   181→   181→    a_shift = cur_ap - op_ap
   182→   182→
   183→   183→    if abs(h_shift) < 0.03 and abs(a_shift) < 0.03:
   184→   184→        return None
   185→   185→
   186→   186→    if h_shift > a_shift:
   187→   187→        pick, shift_pct = "HOME", h_shift * 100
   188→   188→    else:
   189→   189→        pick, shift_pct = "AWAY", a_shift * 100
   190→   190→
   191→   191→    raw_conf = min(95, 30 + abs(shift_pct) * 2.5)
   192→   192→
   193→   193→    # Cross-validate with Handicap rating
   194→   194→    rating_boost = 0
   195→   195→    hc = markets.get("Handicap")
   196→   196→    if hc and hc["latest"]:
   197→   197→        for r in hc["latest"].get("rating", []):
   198→   198→            if r and isinstance(r, dict):
   199→   199→                rd = r.get("direction", "")
   200→   200→                rv = abs(r.get("rating", 0))
   201→   201→                if (pick == "HOME" and rd == "Home") or (pick == "AWAY" and rd == "Away"):
   202→   202→                    rating_boost = min(10, rv * 2)
   203→   203→                    break
   204→   204→
   205→   205→    # Also cross-validate with HT 1X2 direction
   206→   206→    ht_ml = markets.get("HT 1X2")
   207→   207→    if ht_ml and ht_ml["latest"]:
   208→   208→        ht_op, ht_cur = ht_ml["opening"], ht_ml["latest"]
   209→   209→        ht_op_h, ht_op_a = ht_op.get("row1"), ht_op.get("row3")
   210→   210→        ht_cur_h, ht_cur_a = ht_cur.get("row1"), ht_cur.get("row3")
   211→   211→        if all([ht_op_h, ht_op_a, ht_cur_h, ht_cur_a]):
   212→   212→            ht_h_shift = implied_prob(ht_cur_h) - implied_prob(ht_op_h)
   213→   213→            ht_a_shift = implied_prob(ht_cur_a) - implied_prob(ht_op_a)
   214→   214→            ht_pick = "HOME" if ht_h_shift > ht_a_shift else "AWAY"
   215→   215→            if ht_pick == pick:
   216→   216→                rating_boost = min(rating_boost + 5, 12)
   217→   217→
   218→   218→    confidence = min(98, raw_conf + rating_boost)
   219→   219→    details = {
   220→   220→        "op_home": op_h, "op_away": op_a,
   221→   221→        "cur_home": cur_h, "cur_away": cur_a,
   222→   222→        "home_shift": round(h_shift * 100, 1),
   223→   223→        "away_shift": round(a_shift * 100, 1),
   224→   224→        "rating_boost": round(rating_boost, 1),
   225→   225→    }
   226→   226→    return (pick, round(confidence, 1), details)

   229→   229→def predict_total(markets, game):
   230→   230→    tot = markets.get("Total")
   231→   231→    if not tot or not tot["opening"] or not tot["latest"]:
   232→   232→        return None
   233→   233→    op, cur = tot["opening"], tot["latest"]
   234→   234→    op_line, cur_line = op.get("row2"), cur.get("row2")
   235→   235→    cur_over, cur_under = cur.get("row1"), cur.get("row3")
   236→   236→    if not all([op_line, cur_line, cur_over, cur_under]):
   237→   237→        return None
   238→   238→
   239→   239→    line_move = cur_line - op_line
   240→   240→    hs, as_ = get_scores(game)
   241→   241→    total_score = hs + as_
   242→   242→    elapsed = get_elapsed_sec(game)
   243→   243→    full_time = tot.get("full_time", 2400)
   244→   244→
   245→   245→    if elapsed < 300:
   246→   246→        return None
   247→   247→
   248→   248→    projected = (total_score / elapsed) * full_time
   249→   249→    pace_gap = projected - cur_line
   250→   250→
   251→   251→    over_prob, under_prob = implied_prob(cur_over), implied_prob(cur_under)
   252→   252→
   253→   253→    score_sig = 1 if pace_gap > 3 else (-1 if pace_gap < -3 else 0)
   254→   254→    line_sig = -1 if line_move < -2 else (1 if line_move > 2 else 0)
   255→   255→    odds_sig = 1 if over_prob > under_prob else -1
   256→   256→    combined = score_sig * 3 + line_sig * 2 + odds_sig * 1
   257→   257→
   258→   258→    if combined == 0:
   259→   259→        return None
   260→   260→
   261→   261→    pick = "OVER" if combined > 0 else "UNDER"
   262→   262→
   263→   263→    raw_conf = 35
   264→   264→    raw_conf += min(25, abs(pace_gap) * 2)
   265→   265→    raw_conf += min(20, abs(line_move) * 3)
   266→   266→    raw_conf += min(10, abs(combined) * 3)
   267→   267→
   268→   268→    rating_boost = 0
   269→   269→    for r in cur.get("rating", []):
   270→   270→        if r and isinstance(r, dict):
   271→   271→            rd = r.get("direction", "")
   272→   272→            if (pick == "OVER" and rd == "Over") or (pick == "UNDER" and rd == "Under"):
   273→   273→                rating_boost = min(10, abs(r.get("rating", 0)) * 3)
   274→   274→                break
   275→   275→            else:
   276→   276→                rating_boost = -5
   277→   277→
   278→   278→    # Cross-validate with HT Total direction
   279→   279→    ht_tot = markets.get("HT Total")
   280→   280→    if ht_tot and ht_tot["latest"]:
   281→   281→        ht_op_line = ht_tot["opening"].get("row2") if ht_tot["opening"] else None
   282→   282→        ht_cur_line = ht_tot["latest"].get("row2")
   283→   283→        if ht_op_line and ht_cur_line:
   284→   284→            ht_line_move = ht_cur_line - ht_op_line
   285→   285→            if (pick == "OVER" and ht_line_move > 1) or (pick == "UNDER" and ht_line_move < -1):
   286→   286→                rating_boost = min(rating_boost + 4, 12)
   287→   287→
   288→   288→    confidence = max(30, min(95, raw_conf + rating_boost))
   289→   289→    details = {
   290→   290→        "op_line": op_line, "cur_line": cur_line,
   291→   291→        "line_move": round(line_move, 1),
   292→   292→        "current_score": total_score,
   293→   293→        "elapsed_min": round(elapsed / 60, 1),
   294→   294→        "projected_total": round(projected, 1),
   295→   295→        "pace_gap": round(pace_gap, 1),
   296→   296→        "cur_over_odds": cur_over, "cur_under_odds": cur_under,
   297→   297→        "rating_boost": rating_boost,
   298→   298→    }
   299→   299→    return (pick, round(confidence, 1), cur_line, details)

   303→   303→    markets = parse_markets(odds_data)
   304→   304→    preds = []

   306→   306→        return preds

   314→   314→    tot = predict_total(markets, game)

   325→   325→def fmt_pred(game, pred, rank=1):
   326→   326→    home = game["home"]["name"]
   327→   327→    away = game["away"]["name"]
   328→   328→    league = game["league"]["name"]
   329→   329→    icon = "\U0001f525" if rank == 1 else "\u26a1"
   330→   330→    lines = [
   331→   331→        f"{icon} #{rank} {pred['type']}",
   332→   332→        f"{home} vs {away}",
   333→   333→        f"{league} | {fmt_time(game)} | {game.get('scores', '0-0')}",
   334→   334→        f"Confidence: {pred['confidence']}%",

   336→   336→    ]

   339→   339→        lines.append(f"Pick: <b>{pred['pick']}</b>")

   345→   345→        rb = round(d['rating_boost'], 1)
   346→   346→        if rb > 0:
   347→   347→            lines.append(f"Rating confirm: +{rb}%")
   348→   348→    else:
   349→   349→        lines.append(f"Pick: <b>{pred['pick']} {pred['line']}</b>")
   350→   350→        lines.append(f"Line: {d['op_line']} -> {d['cur_line']} ({d['line_move']:+.1f})")
   351→   351→        lines.append(f"Score: {d['current_score']}pts in {d['elapsed_min']}min")
   352→   352→        lines.append(f"Pace: {d['projected_total']}pts (gap {d['pace_gap']:+.1f})")
   353→   353→        lines.append(f"Odds: Over {d['cur_over_odds']} / Under {d['cur_under_odds']}")
   354→   354→        rb = round(d['rating_boost'], 1)
   355→   355→        if rb > 0:
   356→   356→            lines.append(f"Rating confirm: +{rb}%")
   357→   357→        elif rb < 0:
   358→   358→            lines.append(f"Rating disagree: {rb}%")

   422→   422→def scan_and_predict():
   423→   423→    global scan_count, pred_count
   424→   424→    games = fetch_live_games()
   425→   425→    if not games:
   426→   426→        return None, None, [], games
   427→   427→    scan_count += 1
   428→   428→    eligible = [g for g in games if get_quarter(g) >= 2]
   429→   429→    if not eligible:
   430→   430→        return None, None, [], games
   431→   431→
   432→   432→    results = []
   433→   433→    for g in eligible:
   434→   434→        try:
   435→   435→            odds = fetch_odds(g["id"])
   436→   436→            if not odds:
   437→   437→                continue
   438→   438→            preds, markets = analyze_game(g, odds)
   439→   439→            for p in preds:
   440→   440→                results.append((g, p))
   441→   441→                pred_count += 1
   442→   442→        except Exception:
   443→   443→            pass
   444→   444→
   445→   445→    results.sort(key=lambda x: x[1]["confidence"], reverse=True)
   446→   446→    if results:
   447→   447→        return results[0][0], results[0][1], results, games
   448→   448→    return None, None, [], games

   452→   452→def auto_scan_worker():
   453→   453→    """Fully automatic background scanner. Runs forever, pushes predictions."""
   454→   454→    global auto_tracking, known_games
   455→   455→    auto_tracking = True
   456→   456→    cycle = 0
   457→   457→

   459→   459→        cycle += 1

   461→   461→            bg, bp, all_results, games = scan_and_predict()
   462→   462→
   463→   463→            # --- New game alerts: notify when games enter Q2 ---

   465→   465→                current_ids = {g["id"] for g in games if get_quarter(g) >= 2}
   466→   466→                new_ids = current_ids - known_games

   475→   475→                            parse_mode="HTML"
   476→   476→                        )
   477→   477→                known_games = current_ids
   478→   478→
   479→   479→            # --- Send predictions ---
   480→   480→            if bg and bp:
   481→   481→                if ALL_PREDICTIONS and all_results:
   482→   482→                    for g, p in all_results:
   483→   483→                        key = f"{g['id']}_{p['type']}"
   484→   484→                        old = last_predictions.get(key, 0)
   485→   485→                        new_conf = p["confidence"]
   486→   486→                        if abs(new_conf - old) >= 5:
   487→   487→                            rank = 1 if (g, p) == (bg, bp) else 2
   488→   488→                            send_pred_with_button(g, p, rank=rank)
   489→   489→                            last_predictions[key] = new_conf
   490→   490→                else:
   491→   491→                    key = f"{bg['id']}_{bp['type']}"

   497→   497→
   498→   498→            # Periodic status every 10 cycles (10 min)
   499→   499→            if cycle % 10 == 0 and games:
   500→   500→                eligible = len([g for g in games if get_quarter(g) >= 2])
   501→   501→                bot.send_message(CHAT_ID,
   502→   502→                    f"\U0001f4ca Scan #{scan_count} | {len(games)} live | {eligible} analyzing | {pred_count} predictions sent",
   503→   503→                    parse_mode="HTML"
   504→   504→                )
   505→   505→
   506→   506→        except Exception as e:
   507→   507→            print(f'Cycle {cycle} error: {e}')
   508→   508→
   509→   509→        # Sleep in 1-second increments for instant stop capability
   510→   510→        for _ in range(SCAN_INTERVAL):
   511→   511→            if not auto_tracking:
   512→   512→                break
   513→   513→            time.sleep(1)
